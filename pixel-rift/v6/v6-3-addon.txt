const _v63AvailableEnemyTypes=availableEnemyTypes;

function v63InitEnemy(e){
  if(!e||e.v63Ready)return;
  e.v63Ready=true;
  e.v63T=rand(1.1,2.4);
  e.v63T2=rand(2.2,4.2);
  if(e.type==='bastion'){
    e.shield=(e.shield||0)+e.maxHp*.65;
    e.maxShield=Math.max(e.maxShield||0,e.shield)
  }
}

const _v63BossSkill=bossSkill;

  _v63BossSkill(e);

const _v63BossUpdate=updateEnemy;
updateEnemy=function(e,dt){
  _v63BossUpdate(e,dt);
  if(!game||!e||e.dead||!e.abyss)return;
  e.v63BossT=(e.v63BossT||1.3)-dt;
  e.v63BossT2=(e.v63BossT2||3.2)-dt;
  const ratio=e.hp/e.maxHp;
  const p=nearestPlayer(e);

  if(ratio<.75&&e.v63Phase<1){
    e.v63Phase=1;e.damage*=1.13;e.speed*=1.08;
    toast(e.cataclysm?'✹ 災厄第二權能：裂界':'◆ 深淵第二型態：侵蝕')
  }
  if(ratio<.5&&e.v63Phase<2){
    e.v63Phase=2;e.damage*=1.14;e.speed*=1.08;
    e.shield=(e.shield||0)+e.maxHp*(e.cataclysm?.13:.08);
    toast(e.cataclysm?'✹ 災厄第三權能：不滅':'◆ 深淵第三型態：再構')
  }
  if(ratio<.25&&e.v63Phase<3){
    e.v63Phase=3;e.damage*=1.18;e.speed*=1.1;
    toast(e.cataclysm?'✹ 災厄最終權能：終焉':'◆ 深淵最終型態：暴走')
  }

  if(e.v63BossT<=0&&p){
    e.v63BossT=e.cataclysm?(e.v63Phase>=3?.72:1.05):(e.v63Phase>=3?1.0:1.45);
    const a=Math.atan2(p.y-e.y,p.x-e.x);
    for(let i=-2;i<=2;i++)addEnemyBullet(e.x,e.y,a+i*.085,e.cataclysm?390:325,e.damage*.31,5,e.cataclysm?'#ff4e86':'#d697ff')
  }
  if(e.v63BossT2<=0){
    e.v63BossT2=e.cataclysm?2.5:3.3;
    if(e.cataclysm){
      const elitePool=['voidblade','doomseer','singularity','bastion'];
      for(let i=0;i<3;i++)game.enemies.push(makeEnemy(elitePool[Math.floor(Math.random()*elitePool.length)],{x:clamp(e.x+rand(-130,130),30,WORLD_W-30),y:clamp(e.y+rand(-130,130),30,WORLD_H-30)},1+game.wave*.035))
    }else{

    }
  }
