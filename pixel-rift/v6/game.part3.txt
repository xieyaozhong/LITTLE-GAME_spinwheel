function hurt(p,amount,ex,ey,source=null){
if(p.dead||p.invuln>0||p.dashing>0)return 0;let reduction=p.armor;
if(p.hero==='knight'&&p.hp/p.maxHp<.45)reduction=Math.min(.75,reduction+p.fortress);
let dealt=amount*(1-reduction);if(p.shield>0){const s=Math.min(p.shield,dealt);p.shield-=s;dealt-=s}
if(dealt>0)p.hp-=dealt;p.invuln=.7;shake=7;flash=.08;
const a=Math.atan2(p.y-ey,p.x-ex);p.x+=Math.cos(a)*10;p.y+=Math.sin(a)*10;
if(source&&p.thorns)damageEnemy(source,amount*p.thorns,p,false);
if(p.hero==='knight')for(const e of game.enemies)if(!e.dead&&d2(p,e)<75*75)damageEnemy(e,p.damage*.7,p,false);
if(source?.type==='leech'&&dealt>0)source.hp=Math.min(source.maxHp,source.hp+dealt*.75);
if(source?.mutant&&source.mutations.some(m=>m.id==='vamp')&&dealt>0)source.hp=Math.min(source.maxHp,source.hp+dealt*.55);
if(p.hp<=0){p.hp=0;p.dead=true;p.revive=0;if(!alivePlayers().length)endGame()}return dealt
}
