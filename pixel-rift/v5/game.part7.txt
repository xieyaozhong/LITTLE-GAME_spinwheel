function rect(a,b,w,h,col){ctx.fillStyle=col;ctx.fillRect(Math.round(a),Math.round(b),Math.round(w),Math.round(h))}
function drawBackground(cx,cy,z){
const m=game?.map||MAPS[mapChoice],l=cx-W/(2*z)-60,r=cx+W/(2*z)+60,t=cy-H/(2*z)-60,b=cy+H/(2*z)+60;ctx.fillStyle=m.bg;ctx.fillRect(l,t,r-l,b-t);const g=48,sx=Math.floor(l/g)*g,sy=Math.floor(t/g)*g;ctx.strokeStyle=m.grid;ctx.lineWidth=1;for(let q=sx;q<r;q+=g){ctx.beginPath();ctx.moveTo(q,t);ctx.lineTo(q,b);ctx.stroke()}for(let q=sy;q<b;q+=g){ctx.beginPath();ctx.moveTo(l,q);ctx.lineTo(r,q);ctx.stroke()}ctx.strokeStyle=m.accent+'55';ctx.lineWidth=4;ctx.strokeRect(0,0,WORLD_W,WORLD_H)
}
